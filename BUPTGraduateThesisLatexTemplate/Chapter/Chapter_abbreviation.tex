\begin{abbreviation}
	\item[UAV] Unmanned Aerial Vehicle \hfill 无人机
	\item[LEO] Low Earth Orbit \hfill 低轨
	\item[PNT] Positioning, Navigation and Timing \hfill 定位、导航与授时
	\item[RSSI] Received Signal Strength Indicator \hfill 接收信号强度指示
	\item[AOA] Angle of Arrival \hfill 到达角
	\item[TOA] Time of Arrival \hfill 到达时间
	\item[TDOA] Time Difference of Arrival \hfill 到达时间差
	\item[PDR] Pedestrian Dead Reckoning \hfill 航位推算
	\item[LOS] Line of Sight \hfill 视距传播
	\item[NLOS] Non-Line of Sight \hfill 非视距传播
	\item[OFDM] Orthogonal Frequency Division Multiplexing \hfill 正交频分复用
	\item[MDP] Markov Decision Process \hfill 马尔可夫决策过程
	\item[DQN] Deep Q-Network \hfill 深度 Q 网络
	\item[RL-IFA] Reinforcement Learning-based Information Fusion Algorithm \hfill 基于强化学习的信息融合算法
	\item[EKF] Extended Kalman Filter \hfill 扩展卡尔曼滤波
	\item[REM] Radio Environment Map \hfill 无线电环境地图
	\item[CKM] Channel Knowledge Map \hfill 信道知识地图
	\item[PA-DQN] Physics-Aware Deep Q-Network \hfill 物理感知深度 Q 网络
	\item[GAT] Graph Attention Network \hfill 图注意力网络
	\item[KD] Knowledge Distillation \hfill 知识蒸馏
\end{abbreviation}
